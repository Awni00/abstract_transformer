\section{Transformers: The Sensory and the relational}
\begin{frame}{How to imbue Transformers with explicit relational inductive biases}

    Strength of Transformers: \alert{\emph{attention}}

    Versatile \alert{information retrieval} mechanism

    Composable in \alert{\textit{circuits}} to carry out complex computation\newline
    \tiny{(which we're now beginning to understand through systematic (mechanistic) interpretability work)}
\end{frame}

\begin{frame}{The Transformer architecture, essentially}
    Iterate two basic operations:
    \begin{enumerate}
        \item \alert{Information Retrieval}: Attention
        \[x_i \gets \sum_{j} \alpha_{ij} \, \phi_v(x_j)\]
        \item \alert{Local Processing}: Token-wise MLP
        \[x_i \gets \mathrm{MLP}(x_i)\]
    \end{enumerate}
\end{frame}

% manim animation: two red vectors, phi_q, phi_k enter <., .> and leave as a blue relation
\begin{frame}{Attention, essentially}
    \begin{enumerate}
        \item Compute attention scores
        \[\attncolor{\alpha_{ij}} = \mathrm{Softmax}([\iprod{\phiqattn(\senscolor{x_i})}{\phikattn(\senscolor{x_j})}]_{j=1}^{n})_j\]
        \item Retrieve weighted combination of sensory values in context
        \[\senscolor{\bm{e}_i} \gets \sum_{j} \attncolor{\alpha_{ij}} \, \senscolor{\phi_v(x_j)}\]
    \end{enumerate}
\end{frame}

\begin{frame}{Two types of information}
    % \begin{itemize}
    Fundamentally, attention is an information retrieval operation

    Two key \textit{types} of information

    \quad \alert{Sensory}: features or attributes of individual objects

    \quad \alert{Relational}: relationships between objects
    % \end{itemize}

    \emph{Standard attention captures the former, but not the latter}
\end{frame}

\begin{frame}{Two types of attention}
    Correspondingly, their ought to be two types of attention

    % \begin{itemize}
    \alert{(Standard) Sensory Attention}:

    \quad retrieval of \emph{sensory} information in context

    \alert{\textbf{\emph{Relational Attention}}}:

    \quad retrieval of \emph{relational} information in context
    % \end{itemize}
\end{frame}

\section{Relational Attention}
\begin{frame}{High-level: Relational Attention}
    % high-level : 1) attend; 2) relate; 3) tag with symbols
    \Large{
    \begin{enumerate}
        \item Attend
        \item Relate
        \item Tag with symbols
    \end{enumerate}
    }
\end{frame}

\begin{frame}{Attention}
    Same as standard (sensory attention)

    Compute attention scores via learned query/key maps

    \[\attncolor{\alpha_{ij}} = \mathrm{Softmax}([\iprod{\phiqattn(\senscolor{x_i})}{\phikattn(\senscolor{x_j})}]_{j=1}^{n})_j\]
\end{frame}

\begin{frame}{Computing Relations}
    Relation vector, representing a series of comparisons under different attributes or extracted features

    Computed as a series inner products under learned faeture maps

    \[\relcolor{\bm{r}_{ij}} = \paren{\iprod{\phiqrell{\ell}(\senscolor{x_i})}{\phikrell{\ell}(\senscolor{x_j})}}_{\ell \in [d_r]} \in \reals^{d_r}\]
\end{frame}

\begin{frame}{Symbols (High-Level)}
    Tag each object in the context with an \symbcolor{\emph{symbol}}

    \[(\symbcolor{s_1}, \ldots, \symbcolor{s_n}) = \SymbolRetriever(\senscolor{x_1}, \ldots, \senscolor{x_n})\]

    Serve as reference/pointer/identifier of selected object with whom the relation is with, abstracted away from high-dimensional sensory features

    We experiment with different symbol assignment mechanisms: 
\end{frame}

\begin{frame}{Relational Attention: Putting it all together}
    \emph{Putting it all together}
    \begin{align*}
        % \RelAttn(x, (\ys)) = ...
        \relcolor{\bm{a}_i} \gets \sum_{j} \attncolor{\alpha_{ij}} \cdot \paren{W_r \, \relcolor{\bm{r}_{ij}} + W_s \, \symbcolor{s_{j}}}
    \end{align*}

    $W_r, W_s$ are learned linear maps: organize information in residual stream
\end{frame}

\begin{frame}{A few comments...}
    Causal masking

    Positional encoding

    Symmetric relations

    Computational complexity
\end{frame}

\section{Dual Attention Transformer Architecture}

\begin{frame}{Dual Attention Transformer}
    Both \senscolor{\emph{sensory}} and \emph{\relcolor{relational}} information are crucial for reasoning over collections or sequences of objects.

    \alert{\emph{Dual Attention Transformer (DAT)}}: A variant of the Transformer architecture that routes both types of information in the information retrieval step.

    Introduces explicit relational processing mechanisms, while retaining sensory processing capabilities.
\end{frame}

\begin{frame}{Dual Attention}
    \emph{\alert{Dual Attention}} is a variant of multi-head attention with \emph{\alert{two types of attention heads}}:  \senscolor{\emph{sensory}} and \emph{\relcolor{relational}}.
\end{frame}

\begin{frame}{Dual Attention}
    \centering
    \scalebox{0.65}{
    \begin{algorithm}[H]
        \caption{Dual Attention}%\label{alg:dual_head_attn}
        \SetKwInput{Hyperparameters}{Hyperparameters}
        \SetKwInput{Input}{Input}
        \SetKwInOut{Output}{Output}
        % \Hyperparameters{$\nhsa, \nhra$, symbol assignment mechanism, (optional: \texttt{symmetric\_RA = False}, $d_r = \nhra$, ...)}
        \Input{$\bm{x} = (x_1, \ldots, x_n) \in \reals^{n \times d}$ }

        \vspace{1em}

        \texttt{Compute self-attention heads}
        \begin{align*}
            \attncolor{\bm{\alpha}^{(h)}} &\gets \Softmax\bigparen{{(\bm{x} \, \Wqattnn{h})} {(\bm{x} \, \Wkattnn{h})}^\intercal}, \quad h \in [\nhsa] \\
            \senscolor{e_i^{(h)}} &\gets \sum_{j} \attncolor{\alpha_{ij}^{(h)}} \senscolor{x_j} \, W_v^h, \quad i \in [n], h \in [\nhsa]\\
            \senscolor{\bm{e}_i} &\gets \concat\bigparen{e_i^{(1)}, \ldots, e_i^{(\nhsa)}} \, W_o^{sa}, \quad i \in [n]
        \end{align*}

        \texttt{Assign symbols:} $\s = (s_1, \ldots, s_n) \gets \SymbolRetriever(\bm{x};\,\Slib)$
        \vspace{0.5em}

        \texttt{Compute relational attention heads}
        \begin{align*}
            \attncolor{\bm{\alpha}^{(h)}} &\gets \Softmax\bigparen{{(\bm{x} \, \Wqattnn{h})} {(\bm{x} \, \Wkattnn{h})}^\intercal}, \quad h \in [\nhra] \\
            \relcolor{\bm{r}_{ij}} &\gets \bigparen{\iiprod{x_i \, \Wqrell{\ell}}{x_j \, \Wkrell{\ell}}}_{\ell \in [d_r]} \quad i,j \in [n] \\
            \relcolor{\bm{a}_i^{(h)}} &\gets \sum_{j} \attncolor{\alpha_{ij}^{(h)}} \bigparen{\relcolor{\bm{r}_{ij}}\,W_r^h + \symbcolor{s_j} \, W_s^{h}}, \quad i \in [n],\, h \in [\nhra]\\
            \relcolor{\bm{a}_i} &\gets \concat\bigparen{\relcolor{a_i^{(1)}}, \ldots, \relcolor{a_i^{(\nhra)}}} W_o^{ra}, \quad i \in [n]
        \end{align*}

        \Output{$\bigparen{\concat(e_i, a_i)}_{i=1}^{n}$}

    \end{algorithm}
    }
\end{frame}
